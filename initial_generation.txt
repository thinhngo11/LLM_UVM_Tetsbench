// DUT: 4-bit Up/Down Counter
module counter (
    input  logic        clk,
    input  logic        rst_n,
    input  logic        en,
    input  logic        up_dn,
    input  logic        load,
    input  logic [3:0]  data_in,
    output logic [3:0]  count
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            count <= 4'b0;
        else if (load)
            count <= data_in;
        else if (en) begin
            if (up_dn)
                count <= count + 1;
            else
                count <= count - 1;
        end
    end
endmodule

// Interface
interface counter_if (
    input logic clk
);
    logic        rst_n;
    logic        en;
    logic        up_dn;
    logic        load;
    logic [3:0]  data_in;
    logic [3:0]  count;

    clocking cb @(posedge clk);
        input  count;
        output rst_n, en, up_dn, load, data_in;
    endclocking

    modport dut_mp (
        input  clk, rst_n, en, up_dn, load, data_in,
        output count
    );

    // Assertions
    // Reset assertion: count should be 0 after reset
    property p_reset;
        @(posedge clk) !rst_n |=> (count == 4'b0);
    endproperty
    assert_reset: assert property (p_reset)
        else $error("Reset assertion failed: count != 0 after reset");

    // Load assertion: count should equal data_in when load is active
    property p_load;
        @(posedge clk) disable iff (!rst_n)
        load |=> (count == $past(data_in));
    endproperty
    assert_load: assert property (p_load)
        else $error("Load assertion failed: count != data_in after load");

    // Enable assertion: count should change only when en is high (unless load)
    property p_enable;
        @(posedge clk) disable iff (!rst_n || load)
        !en |=> (count == $past(count));
    endproperty
    assert_enable: assert property (p_enable)
        else $error("Enable assertion failed: count changed when en=0");
endinterface

// Package for UVM components
package counter_pkg;
    import uvm_pkg::*;
    `include "uvm_macros.svh"

    // Transaction
    class counter_trans extends uvm_sequence_item;
        rand bit        rst_n;
        rand bit        en;
        rand bit        up_dn;
        rand bit        load;
        rand bit [3:0]  data_in;
        bit [3:0]       count;

        `uvm_object_utils_begin(counter_trans)
            `uvm_field_int(rst_n, UVM_ALL_ON)
            `uvm_field_int(en, UVM_ALL_ON)
            `uvm_field_int(up_dn, UVM_ALL_ON)
            `uvm_field_int(load, UVM_ALL_ON)
            `uvm_field_int(data_in, UVM_ALL_ON)
            `uvm_field_int(count, UVM_ALL_ON)
        `uvm_object_utils_end

        function new(string name = "counter_trans");
            super.new(name);
        endfunction

        // Constraints
        constraint c_rst_n { rst_n dist {1:/90, 0:/10}; } // Reset 10% of time
        constraint c_load { load dist {0:/80, 1:/20}; }   // Load 20% of time
        constraint c_en { en dist {1:/70, 0:/30}; }       // Enable 70% of time
    endclass

    // Sequence
    class counter_seq extends uvm_sequence #(counter_trans);
        `uvm_object_utils(counter_seq)

        function new(string name = "counter_seq");
            super.new(name);
        endfunction

        task body();
            counter_trans trans;
            repeat (100) begin
                trans = counter_trans::type_id::create("trans");
                start_item(trans);
                if (!trans.randomize()) `uvm_error("SEQ", "Randomization failed")
                finish_item(trans);
            end
        endtask
    endclass

    // Driver
    class counter_driver extends uvm_driver #(counter_trans);
        virtual counter_if vif;
        `uvm_component_utils(counter_driver)

        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction

        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual counter_if)::get(this, "", "vif", vif))
                `uvm_fatal("DRV", "No virtual interface set")
        endfunction

        task run_phase(uvm_phase phase);
            super.run_phase(phase);
            forever begin
                counter_trans trans;
                @(vif.cb);
                seq_item_port.get_next_item(trans);
                vif.cb.rst_n   <= trans.rst_n;
                vif.cb.en      <= trans.en;
                vif.cb.up_dn   <= trans.up_dn;
                vif.cb.load    <= trans.load;
                vif.cb.data_in <= trans.data_in;
                seq_item_port.item_done();
            end
        endtask
    endclass

    // Monitor
    class counter_monitor extends uvm_monitor;
        virtual counter_if vif;
        uvm_analysis_port #(counter_trans) ap;

        `uvm_component_utils(counter_monitor)

        function new(string name, uvm_component parent);
            super.new(name, parent);
            ap = new("ap", this);
        endfunction

        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual counter_if)::get(this, "", "vif", vif))
                `uvm_fatal("MON", "No virtual interface set")
        endfunction

        task run_phase(uvm_phase phase);
            super.run_phase(phase);
            forever begin
                counter_trans trans;
                @(vif.cb);
                trans = counter_trans::type_id::create("trans");
                // Sample inputs directly from interface, output via clocking block
                trans.rst_n   = vif.rst_n;
                trans.en      = vif.en;
                trans.up_dn   = vif.up_dn;
                trans.load    = vif.load;
                trans.data_in = vif.data_in;
                trans.count   = vif.cb.count;
                ap.write(trans);
              `uvm_info("MON", $sformatf("vif.cb.rst_n=%0b vif.cb.en=%b, vif.cb.up_dn=%b, vif.cb.load=%b, vif.cb.data_in=%0h, vif.cb.count=%0h", vif.rst_n, vif.en, vif.up_dn, vif.load, vif.data_in, vif.cb.count), UVM_MEDIUM)            
            end
        endtask
    endclass
          
    // Scoreboard
    class counter_scoreboard extends uvm_scoreboard;
        uvm_analysis_imp #(counter_trans, counter_scoreboard) ap;
        bit [3:0] exp_count;

        `uvm_component_utils(counter_scoreboard)

        function new(string name, uvm_component parent);
            super.new(name, parent);
            ap = new("ap", this);
            exp_count = 4'b0;
        endfunction

        function void write(counter_trans trans);
            bit [3:0] act_count = trans.count;
              `uvm_info("MON", $sformatf("trans.rst_n=%0b trans.en=%b, trans.up_dn=%b, trans.load=%b, trans.data_in=%0h, trans.count=%0h", trans.rst_n, trans.en, trans.up_dn, trans.load, trans.data_in, trans.count), UVM_MEDIUM);
            if (!trans.rst_n) begin
                exp_count = 4'b0;
            end
            else if (trans.load) begin
                exp_count = trans.data_in;
            end
            else if (trans.en) begin
                if (trans.up_dn)
                    exp_count = exp_count + 1;
                else
                    exp_count = exp_count - 1;
            end

            if (act_count !== exp_count) begin
                `uvm_error("SCB", $sformatf("Mismatch: Expected count=%0h, Actual count=%0h", exp_count, act_count))
            end
            else begin
                `uvm_info("SCB", $sformatf("Match: Count=%0h", act_count), UVM_MEDIUM)
            end
        endfunction
    endclass

    // Coverage
    class counter_coverage extends uvm_subscriber #(counter_trans);
        `uvm_component_utils(counter_coverage)

        covergroup cg_counter;
            option.per_instance = 1;
            cp_rst_n: coverpoint trans.rst_n;
            cp_en: coverpoint trans.en;
            cp_up_dn: coverpoint trans.up_dn;
            cp_load: coverpoint trans.load;
            cp_data_in: coverpoint trans.data_in {
                bins low  = {0, 1, 2, 3};
                bins mid  = {4, 5, 6, 7, 8, 9, 10, 11};
                bins high = {12, 13, 14, 15};
            }
            cp_count: coverpoint trans.count {
                bins low  = {0, 1, 2, 3};
                bins mid  = {4, 5, 6, 7, 8, 9, 10, 11};
                bins high = {12, 13, 14, 15};
            }
            cross_rst_en: cross cp_rst_n, cp_en;
            cross_load_data: cross cp_load, cp_data_in;
            cross_en_up_dn: cross cp_en, cp_up_dn;
        endgroup

        counter_trans trans;

        function new(string name, uvm_component parent);
            super.new(name, parent);
            cg_counter = new();
            trans = new();
        endfunction

        function void write(counter_trans t);
            trans = t;
            cg_counter.sample();
        endfunction

        function void report_phase(uvm_phase phase);
            super.report_phase(phase);
            `uvm_info("COV", $sformatf("Coverage: %0.2f%%", cg_counter.get_coverage()), UVM_LOW)
        endfunction
    endclass

    // Agent
    class counter_agent extends uvm_agent;
        counter_driver    driver;
        counter_monitor   monitor;
        uvm_sequencer #(counter_trans) sequencer;

        `uvm_component_utils(counter_agent)

        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction

        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            monitor = counter_monitor::type_id::create("monitor", this);
            if (get_is_active()) begin
                driver = counter_driver::type_id::create("driver", this);
                sequencer = uvm_sequencer#(counter_trans)::type_id::create("sequencer", this);
            end
        endfunction

        function void connect_phase(uvm_phase phase);
            super.connect_phase(phase);
            if (get_is_active())
                driver.seq_item_port.connect(sequencer.seq_item_export);
        endfunction
    endclass

    // Environment
    class counter_env extends uvm_env;
        counter_agent      agent;
        counter_scoreboard scoreboard;
        counter_coverage   coverage;

        `uvm_component_utils(counter_env)

        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction

        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            agent = counter_agent::type_id::create("agent", this);
            scoreboard = counter_scoreboard::type_id::create("scoreboard", this);
            coverage = counter_coverage::type_id::create("coverage", this);
        endfunction

        function void connect_phase(uvm_phase phase);
            super.connect_phase(phase);
            agent.monitor.ap.connect(scoreboard.ap);
            agent.monitor.ap.connect(coverage.analysis_export);
        endfunction
    endclass

    // Test
    class counter_test extends uvm_test;
        counter_env env;
        `uvm_component_utils(counter_test)

        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction

        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            env = counter_env::type_id::create("env", this);
        endfunction

        task run_phase(uvm_phase phase);
            counter_seq seq;
            phase.raise_objection(this);
            seq = counter_seq::type_id::create("seq");
            seq.start(env.agent.sequencer);
            phase.drop_objection(this);
        endtask
    endclass
endpackage

module tb;
    import uvm_pkg::*;
    import counter_pkg::*;

    logic clk;

    // Clock generation
    initial begin
        clk = 0;
        forever #5 clk = ~clk;
    end

    // Interface and DUT instantiation
    counter_if cif (.clk(clk));
    counter dut (
        .clk    (cif.clk),
        .rst_n  (cif.rst_n),
        .en     (cif.en),
        .up_dn  (cif.up_dn),
        .load   (cif.load),
        .data_in(cif.data_in),
        .count  (cif.count)
    );

    initial begin
       // Dump file for waveform viewing
        $dumpfile("counter_wave.vcd");
        $dumpvars(0, tb); // Dump all signals in tb hierarchy
        uvm_config_db#(virtual counter_if)::set(null, "*", "vif", cif);
        run_test("counter_test");
    end
endmodule
